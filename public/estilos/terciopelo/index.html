import { Buzon } from 'buzon';

const buzon = new Buzon('bz_123456');

await buzon.whatsapp.enviar({
  para: '+57 300 000 0000',
  plantilla: 'recordatorio_cita',
  datos: { hora: '3:00 p. m.', sede: 'Chapinero' },
});
